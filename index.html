2. Факторіал ($n!$) та числа Фібоначчі

$n! = 1 \cdot 2 \cdot \dots \cdot n$ ($0! = 1$). Для $n \le 20$ використовуємо unsigned long long. Послідовність Фібоначчі: $F_1 = 1, F_2 = 1, F_i = F_{i-1} + F_{i-2}$.

#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;
    // Факторіал
    unsigned long long fact = 1;
    for (int i = 2; i <= n; ++i) fact *= i;
    cout << "n! = " << fact << endl;

    // n-те число Фібоначчі
    unsigned long long a = 1, b = 1, c = 1;
    for (int i = 3; i <= n; ++i) {
        c = a + b;
        a = b;
        b = c;
    }
    cout << "F(n) = " << (n == 0 ? 0 : c) << endl;
    return 0;
}

2. Гіпотеза Коллатца (проблема $3n + 1$)

Візьмемо будь-яке натуральне число $n > 1$. Якщо $n$ парне — ділимо на $2$; якщо непарне — множимо на $3$ і додаємо $1$. Гіпотеза стверджує, що послідовність завжди дійде до $1$.

long long n;
cin >> n;
int steps = 0;
while (n != 1) {
    if (n % 2 == 0) n /= 2;
    else n = 3 * n + 1;
    steps++;
}
cout << "Кроків: " << steps << endl;

#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int n;
    cin >> n;
    int mid = n / 2;
    for (int i = 0; i < n; ++i) {
        int spaces = mid - abs(mid - i);
        int stars = n - 2 * spaces;
        for (int s = 0; s < spaces; ++s) cout << ' ';
        for (int k = 0; k < stars; ++k) cout << '*';
        cout << '
';
    }
    return 0;
}


        
Задача 3. Рекордсмен Коллатца на відрізку $[A; B]$

        
#include <iostream>
using namespace std;

int main() {
    long long a, b;
    cin >> a >> b;
    long long bestNum = a;
    int maxSteps = -1;

    for (long long x = a; x <= b; ++x) {
        long long cur = x;
        int steps = 0;
        while (cur != 1) {
            if (cur % 2 == 0) cur /= 2;
            else cur = 3 * cur + 1;
            ++steps;
        }
        if (steps > maxSteps) {
            maxSteps = steps;
            bestNum = x;
        }
    }
    cout << bestNum << " " << maxSteps << endl;
    return 0;
}
